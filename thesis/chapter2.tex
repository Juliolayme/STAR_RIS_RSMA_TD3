\chapter{MÔ HÌNH HỆ THỐNG VÀ BÀI TOÁN TỐI ƯU}
\label{chap:system-model}

\section{Kiến trúc hệ thống}
Luận văn xét đường xuống của một tế bào đơn gồm một trạm gốc (Base Station, BS) SISO, một STAR--RIS thụ động có $N$ phần tử và $K=4$ người dùng SISO. Người dùng 1 và 3 nằm ở miền phản xạ; người dùng 2 và 4 nằm ở miền truyền qua. Việc gán phía được cố định trong toàn bộ kịch bản để tập trung đánh giá bài toán phân bổ tài nguyên.

BS phát một luồng chung và bốn luồng riêng theo RSMA. Mỗi người dùng nhận tín hiệu từ đường trực tiếp BS--người dùng và đường ghép tầng BS--STAR--RIS--người dùng. STAR--RIS vận hành theo ES, vì vậy mỗi phần tử đồng thời tạo thành phần truyền qua và phản xạ.

Hình~\ref{fig:ch2-system-model} thể hiện STAR--RIS dưới dạng tấm nghiêng theo phối cảnh. Miền phản xạ nằm bên trái, miền truyền qua nằm bên phải. Đường xám nét đứt biểu diễn kênh trực tiếp; đường xanh dương từ BS đến STAR--RIS cùng các đường xanh lá hoặc cam tạo thành kênh ghép tầng. Hình cũng cho thấy một luồng chung $s_c$ và bốn luồng riêng $s_1,\ldots,s_4$ được truyền đồng thời.

\section{Các giả định nghiên cứu}
Các giả định được sử dụng nhất quán trong source code và mô phỏng gồm:
\begin{itemize}
  \item CSI hoàn hảo được cung cấp cho bộ ra quyết định;
  \item các hệ số kênh được sinh độc lập theo phân bố Gaussian phức;
  \item STAR--RIS thụ động, vận hành theo ES và không khuếch đại tín hiệu;
  \item pha truyền và pha phản xạ được điều khiển độc lập;
  \item SIC lý tưởng, không có nhiễu dư sau khi loại bỏ luồng chung;
  \item mỗi bước sử dụng một kịch bản kênh độc lập; hành động không làm thay đổi phân bố kênh kế tiếp;
  \item không xét chi phí ước lượng CSI, báo hiệu điều khiển và tiêu thụ công suất phần cứng.
\end{itemize}

Các giả định này giúp cô lập tác động của thuật toán nhưng giới hạn khả năng suy rộng sang hệ thống thực. Đặc biệt, mô hình pha độc lập có thể không phù hợp với một số kiến trúc STAR--RIS thụ động có ràng buộc ghép pha \cite{Liu2021STARCoupledPhase,Wang2022STAROptimization}.

\section{Mô hình kênh}
\subsection{Kênh trực tiếp và kênh ghép tầng}
Gọi $h_{d,k}\in\mathbb{C}$ là kênh trực tiếp từ BS đến người dùng $k$, $\bm g\in\mathbb{C}^{N}$ là kênh từ BS đến STAR--RIS và $\bm h_{r,k}\in\mathbb{C}^{N}$ là kênh từ STAR--RIS đến người dùng $k$. Trong bộ sinh kênh của luận văn,
\begin{align}
 h_{d,k}&\sim\mathcal{CN}(0,0{,}35^2),\\
 g_n&\sim\mathcal{CN}(0,1^2),\\
 h_{r,k,n}&\sim\mathcal{CN}(0,0{,}75^2),
 \label{eq:channel-distributions}
\end{align}
độc lập giữa các hệ số và giữa các kịch bản.

Nhiễu tại người dùng được mô hình bằng nhiễu Gaussian trắng cộng (Additive White Gaussian Noise, AWGN)
\begin{equation}
 n_k\sim\mathcal{CN}(0,\sigma^2),
\end{equation}
với $\sigma^2=0{,}001$ trong cấu hình mô phỏng.

Các hệ số tỷ lệ trong \eqref{eq:channel-distributions} tạo khác biệt tương đối giữa ba loại kênh nhưng không phải mô hình suy hao dựa trên khoảng cách. Luận văn không gán tọa độ, không dùng hệ số Rician và không mô hình tương quan không gian.

\subsection{Hệ số STAR--RIS}
Với phần tử thứ $n$, hệ số chia năng lượng truyền qua là $\beta_n^T\in[0,1]$ và hệ số phản xạ là
\begin{equation}
 \beta_n^R=1-\beta_n^T.
 \label{eq:beta-conservation-ch2}
\end{equation}
Hai hệ số phức được viết
\begin{align}
 \phi_n^T&=\sqrt{\beta_n^T}\,e^{j\theta_n^T},\\
 \phi_n^R&=\sqrt{1-\beta_n^T}\,e^{j\theta_n^R},
 \label{eq:star-coefficients}
\end{align}
trong đó $\theta_n^T,\theta_n^R\in[-\pi,\pi)$. Dấu căn bậc hai bảo đảm $\beta$ được hiểu là tỷ lệ công suất, còn hệ số nhân vào tín hiệu là biên độ.

Đặt
\begin{align}
 \bm\Phi_T&=\operatorname{diag}(\phi_1^T,\ldots,\phi_N^T),\\
 \bm\Phi_R&=\operatorname{diag}(\phi_1^R,\ldots,\phi_N^R).
\end{align}
Gọi $u_k\in\{0,1\}$ là chỉ báo phía người dùng, với $u_k=1$ cho miền truyền qua và $u_k=0$ cho miền phản xạ. Khi đó
\begin{equation}
 \bm\Phi_{u_k}=u_k\bm\Phi_T+(1-u_k)\bm\Phi_R.
\end{equation}

\subsection{Kênh hiệu dụng}
Kênh hiệu dụng của người dùng $k$ là tổng đường trực tiếp và đường ghép tầng:
\begin{equation}
 h_k^{\mathrm{eff}}=h_{d,k}+\bm h_{r,k}^{H}\bm\Phi_{u_k}\bm g.
 \label{eq:effective-channel-ch2}
\end{equation}
Độ lợi công suất tương ứng là
\begin{equation}
 G_k=\left|h_k^{\mathrm{eff}}\right|^2.
\end{equation}
Một vector pha phải phục vụ nhiều người dùng có kênh khác nhau, vì vậy không thể đồng thời căn chỉnh hoàn hảo cho mọi người dùng. Đây là một nguồn tạo tính phi lồi của bài toán.

\section{Mô hình tín hiệu RSMA}
\subsection{Tín hiệu phát}
BS truyền tín hiệu
\begin{equation}
 x=\sqrt{p_c}s_c+\sum_{k=1}^{K}\sqrt{p_k}s_k,
 \label{eq:rsma-transmit-ch2}
\end{equation}
trong đó $p_c\ge0$, $p_k\ge0$ và
\begin{equation}
 p_c+\sum_{k=1}^{K}p_k\le P_{\max}.
 \label{eq:power-budget-ch2}
\end{equation}
Các ký hiệu dữ liệu được chuẩn hóa có công suất trung bình bằng một và độc lập với nhau.

\ThesisFigure{figures/pdf/chapter2_fig02_rsma_transmit_decode.pdf}
{Quy trình tạo tín hiệu và giải mã luồng chung, luồng riêng trong RSMA}
{fig:ch2-rsma-decode}[0.95\linewidth]
\figuresource{Tác giả xây dựng dựa trên mô hình RSMA một lớp \cite{Mao2022RSMAFundamentals} (2026)}

Hình~\ref{fig:ch2-rsma-decode} mô tả quá trình chia thông điệp, mã hóa và giải mã. Tất cả người dùng đều giải mã $s_c$. Sau SIC, người dùng $k$ chỉ giải mã luồng riêng $s_k$ và xem các luồng riêng khác như nhiễu.

\subsection{Giải mã luồng chung}
Tín hiệu thu tại người dùng $k$ là
\begin{equation}
 y_k=h_k^{\mathrm{eff}}x+n_k.
\end{equation}
Khi giải mã luồng chung, tất cả luồng riêng được xem như nhiễu. SINR của luồng chung là
\begin{equation}
 \gamma_{c,k}=\frac{G_kp_c}{G_k\sum_{j=1}^{K}p_j+\sigma^2}.
 \label{eq:common-sinr-ch2}
\end{equation}
Tốc độ luồng chung mà người dùng $k$ có thể giải mã là
\begin{equation}
 R_{c,k}=\log_2(1+\gamma_{c,k}).
\end{equation}
Do mọi người dùng dùng cùng mã của luồng chung, tốc độ chung của hệ thống bị giới hạn bởi người dùng yếu nhất:
\begin{equation}
 R_c=\min_{k=1,\ldots,K}R_{c,k}.
 \label{eq:common-rate-min-ch2}
\end{equation}

Gọi $\eta_k\ge0$ là tỷ lệ tốc độ chung cấp cho người dùng $k$, với
\begin{equation}
 \sum_{k=1}^{K}\eta_k=1.
 \label{eq:common-simplex-ch2}
\end{equation}
Phần tốc độ chung của người dùng là
\begin{equation}
 C_k=\eta_kR_c.
\end{equation}

\subsection{Giải mã luồng riêng}
Sau khi loại bỏ luồng chung, SINR của luồng riêng $k$ là
\begin{equation}
 \gamma_{p,k}=\frac{G_kp_k}{G_k\sum_{j\ne k}p_j+\sigma^2}.
 \label{eq:private-sinr-ch2}
\end{equation}
Tốc độ riêng và tốc độ cuối của người dùng lần lượt là
\begin{align}
 R_{p,k}&=\log_2(1+\gamma_{p,k}),\\
 R_k&=C_k+R_{p,k}.
 \label{eq:user-rate-ch2}
\end{align}
Tổng tốc độ hệ thống được tính
\begin{equation}
 R_{\mathrm{sum}}=\sum_{k=1}^{K}R_k.
 \label{eq:sum-rate-ch2}
\end{equation}

\section{Chỉ số QoS}
Luận văn sử dụng ba chỉ số để mô tả tính khả thi dịch vụ. Với ngưỡng $R_{\min}=0{,}5$ bit/s/Hz, tỷ lệ người dùng đạt QoS trong một kịch bản là
\begin{equation}
 q_{\mathrm{frac}}=\frac{1}{K}\sum_{k=1}^{K}\mathbb{I}(R_k\ge R_{\min}).
 \label{eq:qos-fraction-ch2}
\end{equation}
Chỉ báo toàn bộ người dùng đạt QoS là
\begin{equation}
 q_{\mathrm{all}}=\mathbb{I}\!\left(\min_k R_k\ge R_{\min}\right).
 \label{eq:all-qos-ch2}
\end{equation}
Mức vi phạm liên tục được tính
\begin{equation}
 V=\sum_{k=1}^{K}\max(R_{\min}-R_k,0).
 \label{eq:violation-ch2}
\end{equation}
Ba chỉ số bổ sung cho nhau. $q_{\mathrm{frac}}$ cho biết tỷ lệ người dùng đạt yêu cầu; $q_{\mathrm{all}}$ nhạy với bất kỳ người dùng vi phạm nào; $V$ phân biệt vi phạm nhẹ và vi phạm nghiêm trọng.

\section{Biến tối ưu và ràng buộc}
Tập biến quyết định gồm
\begin{equation}
 \mathcal{A}=\{\bm p,\bm\eta,\bm\beta^T,\bm\theta^T,\bm\theta^R\},
\end{equation}
trong đó
\begin{align}
 \bm p&=[p_c,p_1,\ldots,p_K]^T,\\
 \bm\eta&=[\eta_1,\ldots,\eta_K]^T,\\
 \bm\beta^T&=[\beta_1^T,\ldots,\beta_N^T]^T,\\
 \bm\theta^T&=[\theta_1^T,\ldots,\theta_N^T]^T,\\
 \bm\theta^R&=[\theta_1^R,\ldots,\theta_N^R]^T.
\end{align}

{Các nhóm biến quyết định, mục tiêu và ràng buộc của bài toán}
{fig:ch2-variables-constraints}[0.94\linewidth]
\figuresource{Tác giả xây dựng từ bài toán tối ưu của luận văn (2026)}

Hình~\ref{fig:ch2-variables-constraints} nhóm các biến theo ý nghĩa vật lý. Công suất và tỷ lệ tốc độ nằm trên simplex; hệ số chia năng lượng bị chặn trong $[0,1]$; hai vector pha nằm trong miền tuần hoàn; tốc độ của mỗi người dùng phải đạt ngưỡng QoS.

Trong implementation, vector công suất được chiếu lên simplex có tổng đúng bằng $P_{\max}$. Vì vậy, policy luôn sử dụng toàn bộ ngân sách công suất. Cách tham số hóa này thu hẹp bài toán tổng quát từ bất đẳng thức \eqref{eq:power-budget-ch2} xuống biên $\sum_i p_i=P_{\max}$.

\section{Phát biểu bài toán tối ưu}
Với một mẫu kênh $\mathcal{H}=\{\bm h_d,\bm g,\bm H_r,\bm u\}$, bài toán được phát biểu
\begin{subequations}
\label{prob:resource-allocation}
\begin{align}
 \max_{\bm p,\bm\eta,\bm\beta^T,\bm\theta^T,\bm\theta^R}
 &\quad R_{\mathrm{sum}}(\mathcal{H},\mathcal{A})\\
 \text{s.t.}
 &\quad \bm p\ge\bm0,\quad \mathbf{1}^T\bm p=P_{\max},\\
 &\quad \bm\eta\ge\bm0,\quad \mathbf{1}^T\bm\eta=1,\\
 &\quad 0\le\beta_n^T\le1,\quad n=1,\ldots,N,\\
 &\quad -\pi\le\theta_n^T,\theta_n^R<\pi,\quad n=1,\ldots,N,\\
 &\quad R_k\ge R_{\min},\quad k=1,\ldots,K.
\end{align}
\end{subequations}

Trong một số kịch bản, miền QoS có thể khó đạt đối với một phương pháp nhất định. Vì vậy, quá trình huấn luyện và các baseline dùng hàm merit mềm
\begin{equation}
 f=R_{\mathrm{sum}}-\lambda_1V-\lambda_2\sum_{k=1}^{K}\max(R_{\min}-R_k,0)^2.
 \label{eq:merit-ch2}
\end{equation}
Các bảng kết quả vẫn báo cáo riêng tổng tốc độ và QoS; reward hoặc merit không được dùng thay thế chỉ số vật lý.

\section{Phân tích tính phi lồi}
Bài toán \eqref{prob:resource-allocation} khó giải do các nguyên nhân chính:
\begin{enumerate}[label=\arabic*)]
  \item kênh hiệu dụng chứa tổng các số phức có pha điều khiển, sau đó được lấy trị tuyệt đối bình phương;
  \item SINR là tỷ số phi tuyến của công suất và độ lợi kênh;
  \item tốc độ chung dùng phép lấy giá trị nhỏ nhất trên người dùng;
  \item các biến công suất, chia năng lượng và pha tác động đồng thời lên nhiều tốc độ;
  \item ràng buộc QoS tạo miền khả thi không lồi và pha là biến tuần hoàn;
  \item số biến tăng theo $2K+1+3N$, nên không gian tìm kiếm mở rộng nhanh khi $N$ tăng.
\end{enumerate}

Đánh giá một action cần tính kênh hiệu dụng cho $K$ người dùng, có độ phức tạp chủ yếu $O(KN)$. Một solver dựa trên sai phân hữu hạn phải lặp lại phép đánh giá này nhiều lần cho từng tọa độ, trong khi Actor đã huấn luyện chỉ cần một lần truyền thẳng và một lần tính metric.

\section{Kết luận chương}
Chương 2 đã xây dựng mô hình SISO STAR--RIS hỗ trợ RSMA, mô tả kênh trực tiếp và kênh ghép tầng, xác định tốc độ chung, tốc độ riêng, các chỉ số QoS và tập biến tối ưu. Bài toán có nhiều biến liên tục, ràng buộc phi lồi và số chiều tăng tuyến tính theo $N$. Chương 3 trình bày cách chuyển bài toán này thành một bộ tối ưu TD3 có hành động luôn hợp lệ về mặt hình học.
