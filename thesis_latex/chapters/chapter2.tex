\chapter{MÔ HÌNH HỆ THỐNG VÀ BÀI TOÁN TỐI ƯU}
\label{chap:mo-hinh-he-thong}

\section{Kiến trúc hệ thống}
\label{sec:kien-truc-he-thong}

Luận văn xét một mạng thông tin di động tế bào đường xuống (downlink) hoạt động ở chế độ một đầu
vào--một đầu ra (SISO). Hệ thống gồm ba thực thể:
\begin{enumerate}
\item \textbf{Trạm gốc (BS)} có duy nhất một anten phát, chịu trách nhiệm xử lý tập trung và phân bổ
  công suất phát.
\item \textbf{STAR--RIS} hoạt động hoàn toàn thụ động, gồm $N$ phần tử ($n \in \{1, \dots, N\}$), được
  đặt ở ranh giới giữa hai khu vực để mở rộng vùng phủ ra cả hai phía bề mặt.
\item \textbf{Các thiết bị người dùng (UE)}: hệ thống phục vụ đồng thời $K = 4$ UE một anten
  ($k \in \{1, 2, 3, 4\}$), chia thành hai nhóm theo vị trí so với bề mặt:
  \begin{itemize}
  \item \textbf{Nhóm phản xạ (R)} nằm cùng phía với BS, nhận tín hiệu phản xạ từ bề mặt; trong mã
    nguồn, chỉ báo phía được gán luân phiên nên nhóm này gồm UE$_1$ và UE$_3$.
  \item \textbf{Nhóm truyền qua (T)} nằm ở phía đối diện, sau bề mặt, nhận tín hiệu truyền xuyên
    qua bề mặt; nhóm này gồm UE$_2$ và UE$_4$.
  \end{itemize}
\end{enumerate}

Tại mỗi thời điểm, BS dùng RSMA một lớp để phát đồng thời dữ liệu cho cả bốn UE. Tín hiệu từ BS
đến mỗi UE đi theo hai đường song song. \textbf{Đường trực tiếp} BS--UE tồn tại với mọi UE, kể cả
các UE ở phía truyền qua. \textbf{Đường ghép tầng} đi từ BS đến STAR--RIS, được $N$ phần tử thụ
động chia biên độ và dịch pha, rồi phản xạ hoặc truyền xuyên qua tới UE. STAR--RIS không khuếch đại
công suất và không giải mã tín hiệu, mà chỉ tái định hình biên độ tương đối và pha của sóng tới; vì
vậy hiệu quả của nó thể hiện qua việc thay đổi kênh hiệu dụng mà RSMA nhìn thấy.
Hình~\ref{fig:ch2-mo-hinh} minh họa mô hình hệ thống và các kênh; đường liền nét là đường ghép tầng
qua bề mặt, đường nét đứt là đường trực tiếp.

\section{Các giả định nghiên cứu}
\label{sec:gia-dinh}

Để mô hình toán học rõ ràng và tập trung vào bài toán phân bổ tài nguyên, luận văn dùng bốn giả
định:
\begin{enumerate}
\item \textbf{CSI hoàn hảo.} BS biết đầy đủ và chính xác kênh tức thời của mọi liên kết tại thời điểm
  ra quyết định. Giả định này tách bài toán phân bổ tài nguyên khỏi bài toán ước lượng kênh.
\item \textbf{SIC lý tưởng.} Sau khi giải mã thành công luồng chung, UE loại hoàn toàn thành phần
  này khỏi tín hiệu thu trước khi giải mã luồng riêng, không có lan truyền lỗi.
\item \textbf{Chia năng lượng thụ động không tổn hao.} Mọi phần tử STAR--RIS hoạt động ở chế độ ES
  và tổng tỷ lệ năng lượng truyền qua và phản xạ của mỗi phần tử luôn bằng một.
\item \textbf{Pha truyền và phản xạ độc lập.} Pha của hai phía được giả định điều khiển độc lập. Đây
  là một giả định lý tưởng hóa: với STAR--RIS thụ động không tổn hao, pha truyền và pha phản xạ của
  cùng một phần tử bị ràng buộc ghép~\cite{Liu2021STARCoupledPhase, Wang2022STAROptimization}.
\end{enumerate}

\section{Mô hình kênh và kênh hiệu dụng}
\label{sec:mo-hinh-kenh}

\subsection{Sinh mẫu kênh phức}
\label{ssec:sinh-mau-kenh}

Kênh được mô hình hóa bằng các biến ngẫu nhiên Gaussian phức đối xứng tròn:
\begin{align}
h_{d,k} &\sim \mathcal{CN}(0, \sigma_d^2), & \sigma_d &= 0,35, \label{eq:kenh-truc-tiep}\\
g_n &\sim \mathcal{CN}(0, \sigma_g^2), & \sigma_g &= 1,0, \label{eq:kenh-bs-ris}\\
h_{r,k,n} &\sim \mathcal{CN}(0, \sigma_r^2), & \sigma_r &= 0,75. \label{eq:kenh-ris-ue}
\end{align}

Trong đó: $h_{d,k} \in \mathbb{C}$ là hệ số kênh trực tiếp từ BS đến UE $k$; $g_n \in \mathbb{C}$ là
hệ số kênh từ BS đến phần tử thứ $n$; $h_{r,k,n} \in \mathbb{C}$ là hệ số kênh từ phần tử thứ $n$
đến UE $k$; $\mathcal{CN}(\mu, \sigma^2)$ là phân bố Gaussian phức đối xứng tròn với trung bình $\mu$
và phương sai $\sigma^2$; $\sigma_d, \sigma_g, \sigma_r$ là các hệ số tỷ lệ (độ lệch chuẩn) của ba
loại liên kết. Mỗi thành phần ghép tầng $h^{*}_{r,k,n} g_n$ có độ lệch chuẩn $\sigma_g \sigma_r =
0,75$, lớn hơn độ lệch chuẩn của đường trực tiếp ($0,35$), phản ánh kịch bản đường trực tiếp yếu.

Đây là các hệ số tỷ lệ của mô phỏng, không phải mô hình suy hao theo khoảng cách. Phương sai của
từng phần tử được giữ cố định khi $N$ thay đổi, nên khi tăng $N$, diện tích hiệu dụng của bề mặt
cũng tăng theo; mức tăng tổng tốc độ theo $N$ trong Chương~\ref{chap:ket-qua} vì vậy không tương ứng
trường hợp giữ cố định kích thước vật lý của bề mặt. Để thuận tiện, các hệ số được gom thành véc-tơ
$\mathbf{g} = [g_1, \dots, g_N]^T \in \mathbb{C}^{N\times 1}$ và
$\mathbf{h}_{r,k} = [h_{r,k,1}, \dots, h_{r,k,N}]^T \in \mathbb{C}^{N\times 1}$.

\subsection{Hệ số truyền và phản xạ của STAR--RIS}
\label{ssec:he-so-star}

Tác động của phần tử thứ $n$ lên sóng truyền qua (T) và sóng phản xạ (R) được mô tả bởi
\begin{equation}
\phi^T_n = \sqrt{\beta^T_n}\,e^{j\theta^T_n}, \qquad
\phi^R_n = \sqrt{1 - \beta^T_n}\,e^{j\theta^R_n}.
\label{eq:he-so-star}
\end{equation}

Trong đó: $\beta^T_n \in [0, 1]$ là tỷ lệ năng lượng truyền qua và $\beta^R_n = 1 - \beta^T_n$ là tỷ
lệ năng lượng phản xạ, nên $\beta^T_n + \beta^R_n = 1$ luôn được thỏa; $\theta^T_n, \theta^R_n \in
[-\pi, \pi)$ là góc dịch pha cho hai phía; $j$ là đơn vị ảo; $e^{j\theta} = \cos\theta + j\sin\theta$.
Ma trận truyền qua và ma trận phản xạ của bề mặt là $\boldsymbol{\Phi}_T = \mathrm{diag}(\phi^T_1,
\dots, \phi^T_N)$ và $\boldsymbol{\Phi}_R = \mathrm{diag}(\phi^R_1, \dots, \phi^R_N)$. Hàm
$\mathrm{wrap}(\cdot)$ được dùng ở Chương~\ref{chap:phuong-phap} để đưa mọi góc pha về miền
$[-\pi, \pi)$.

\subsection{Kênh hiệu dụng}
\label{ssec:kenh-hieu-dung}

Tín hiệu tại UE $k$ là tổng của đường trực tiếp và $N$ thành phần qua STAR--RIS. Kênh hiệu dụng là
\begin{equation}
h^{\mathrm{eff}}_k = h_{d,k} + \mathbf{h}_{r,k}^H \boldsymbol{\Phi}_{s_k} \mathbf{g}
= h_{d,k} + \sum_{n=1}^{N} h^{*}_{r,k,n}\, \phi^{s_k}_n\, g_n .
\label{eq:kenh-hieu-dung}
\end{equation}

Trong đó: $s_k \in \{T, R\}$ là chỉ báo phía của UE $k$ ($s_k = R$ với UE$_1$, UE$_3$ và $s_k = T$ với
UE$_2$, UE$_4$); $\boldsymbol{\Phi}_{s_k}$ và $\phi^{s_k}_n$ là ma trận và hệ số của phía tương ứng;
$(\cdot)^H$ là chuyển vị liên hợp và $(\cdot)^*$ là liên hợp phức. Độ lợi công suất kênh hiệu dụng
được ký hiệu $g_k = |h^{\mathrm{eff}}_k|^2$.

\section{Mô hình tín hiệu RSMA}
\label{sec:mo-hinh-rsma}

\subsection{Tín hiệu phát}
\label{ssec:tin-hieu-phat}

Theo RSMA một lớp, các thông điệp $\{W_1, \dots, W_K\}$ được tách thành một phần chung $W_c$, mã hóa
thành ký hiệu luồng chung $s_c$, và $K$ phần riêng, mã hóa thành các ký hiệu luồng riêng $s_1, \dots,
s_K$. Các ký hiệu độc lập đôi một với công suất trung bình chuẩn hóa $\mathbb{E}[|s_c|^2] =
\mathbb{E}[|s_k|^2] = 1$. Tín hiệu phát của BS một anten là
\begin{equation}
x = \sqrt{p_c}\, s_c + \sum_{k=1}^{K} \sqrt{p_k}\, s_k,
\qquad p_c + \sum_{k=1}^{K} p_k = P_{\max},\quad p_c, p_k \ge 0 .
\label{eq:tin-hieu-phat}
\end{equation}

Trong đó: $p_c$ và $p_k$ là công suất luồng chung và luồng riêng của UE $k$; $P_{\max}$ là tổng công
suất phát (chuẩn hóa bằng $1$ trong thực nghiệm). Vì BS chỉ có một anten, $x$ là đại lượng vô hướng:
luồng chung và mọi luồng riêng đi qua cùng một hệ số kênh $h^{\mathrm{eff}}_k$ để đến UE $k$, không
có véc-tơ tạo búp sóng để tách các luồng như trong hệ MISO hoặc MIMO.

\subsection{Tín hiệu thu}
\label{ssec:tin-hieu-thu}

Tín hiệu băng gốc tại UE $k$ là
\begin{equation}
y_k = h^{\mathrm{eff}}_k \sqrt{p_c}\, s_c + h^{\mathrm{eff}}_k \sqrt{p_k}\, s_k
+ \sum_{j \neq k} h^{\mathrm{eff}}_k \sqrt{p_j}\, s_j + n_k ,
\label{eq:tin-hieu-thu}
\end{equation}

trong đó ba hạng đầu lần lượt là luồng chung, luồng riêng mong muốn và nhiễu từ các luồng riêng của
UE khác, còn $n_k \sim \mathcal{CN}(0, \sigma^2)$ là tạp âm với $\sigma^2 = 10^{-3}$ trong mô phỏng.
Hình~\ref{fig:ch2-kien-truc} tóm tắt kiến trúc thu--phát RSMA một lớp cho $K$ UE trong hệ SISO.

\subsection{Giải mã luồng chung}
\label{ssec:giai-ma-chung}

Ở giai đoạn thứ nhất, mọi UE giải mã luồng chung và xem mọi luồng riêng như nhiễu. SINR và tốc độ
giải mã luồng chung tại UE $k$ là
\begin{equation}
\gamma_{c,k} = \frac{g_k\, p_c}{g_k\, P_{\mathrm{priv}} + \sigma^2},
\qquad R_{c,k} = \log_2(1 + \gamma_{c,k}),
\label{eq:sinr-chung}
\end{equation}

với $P_{\mathrm{priv}} = \sum_{j=1}^{K} p_j = P_{\max} - p_c$. Để mọi UE giải mã được luồng chung, tốc
độ luồng chung bị giới hạn bởi UE có điều kiện thu kém nhất:
\begin{equation}
R_c = \min_{k} R_{c,k} .
\label{eq:toc-do-chung}
\end{equation}

Tốc độ luồng chung được chia cho các UE theo véc-tơ $\boldsymbol{\eta} = [\eta_1, \dots, \eta_K]^T$:
\begin{equation}
C_k = \eta_k R_c, \qquad \eta_k \ge 0, \quad \sum_{k=1}^{K} \eta_k = 1 ,
\label{eq:chia-toc-do-chung}
\end{equation}

trong đó $\boldsymbol{\eta}$ nằm trên đơn hình xác suất $\Delta_K$ và $C_k$ là phần tốc độ chung dành
cho UE $k$.

\subsection{Khử nhiễu liên tiếp và giải mã luồng riêng}
\label{ssec:sic-giai-ma-rieng}

Sau khi giải mã $s_c$, UE $k$ dùng đúng một tầng SIC để loại luồng chung rồi giải mã luồng riêng của
mình, xem các luồng riêng khác như nhiễu:
\begin{equation}
\gamma_{p,k} = \frac{g_k\, p_k}{g_k\,(P_{\mathrm{priv}} - p_k) + \sigma^2},
\qquad R_{p,k} = \log_2(1 + \gamma_{p,k}) .
\label{eq:sinr-rieng}
\end{equation}

Tốc độ của UE $k$ và tổng tốc độ của hệ thống là
\begin{equation}
R_k = \eta_k R_c + R_{p,k}, \qquad
R_{\mathrm{sum}} = \sum_{k=1}^{K} R_k = R_c + \sum_{k=1}^{K} R_{p,k},
\label{eq:tong-toc-do}
\end{equation}

trong đó đẳng thức cuối dùng $\sum_k \eta_k = 1$.

\section{Chỉ số chất lượng dịch vụ và hàm đánh giá}
\label{sec:qos}

Bên cạnh tổng tốc độ, hệ thống yêu cầu mỗi UE đạt tốc độ tối thiểu $R_{\min} = 0,5$ bit/s/Hz. Mức
thiếu hụt của UE $k$ được đo bằng
\begin{equation}
d_k = \max(R_{\min} - R_k, 0),
\label{eq:thieu-hut}
\end{equation}

với $d_k = 0$ khi UE đạt hoặc vượt ngưỡng và $d_k > 0$ cho biết lượng tốc độ còn thiếu. Mức vi phạm QoS
của toàn mạng được tổng hợp bằng hai đại lượng
\begin{equation}
V = \sum_{k=1}^{K} d_k, \qquad V_2 = \sum_{k=1}^{K} d_k^2 .
\label{eq:vi-pham}
\end{equation}

Trong đó: $V$ là tổng thiếu hụt tuyến tính; $V_2$ là tổng bình phương thiếu hụt, phạt nặng hơn khi
một UE riêng lẻ bị thiếu hụt sâu. Từ đó, luận văn định nghĩa hàm đánh giá có phạt QoS dùng chung cho
các phương pháp tối ưu lặp:
\begin{equation}
r_{\mathrm{env}} = R_{\mathrm{sum}} - 8V - 8V_2 .
\label{eq:phan-thuong-moi-truong}
\end{equation}

\section{Tập hợp các biến quyết định}
\label{sec:bien-quyet-dinh}

Tại mỗi lần ra quyết định, bộ điều khiển phải chọn năm nhóm biến liên tục:
\begin{enumerate}
\item công suất luồng chung $p_c \in [0, P_{\max}]$;
\item véc-tơ công suất luồng riêng $\mathbf{p}_{\mathrm{priv}} = [p_1, \dots, p_K]^T \in
  \mathbb{R}_{+}^{K}$;
\item véc-tơ phân bổ tốc độ chung $\boldsymbol{\eta} \in \Delta_K$;
\item véc-tơ chia năng lượng $\boldsymbol{\beta}^T = [\beta^T_1, \dots, \beta^T_N]^T \in [0, 1]^N$,
  với $\beta^R_n = 1 - \beta^T_n$;
\item hai véc-tơ pha $\boldsymbol{\theta}^T, \boldsymbol{\theta}^R \in [-\pi, \pi)^N$.
\end{enumerate}
Tổng số chiều của không gian hành động liên tục là
\begin{equation}
d_a = 1 + K + K + N + N + N = 2K + 1 + 3N .
\label{eq:so-chieu-hanh-dong}
\end{equation}
Với $N = 16$, $d_a = 57$; với $N = 128$, $d_a = 393$ biến liên tục cần quyết định đồng thời, một không
gian tìm kiếm lớn cho cả phương pháp giải tích lẫn học máy.

\section{Phát biểu bài toán tối ưu}
\label{sec:phat-bieu}

Mục tiêu là tìm cấu hình công suất, tỷ lệ chia tốc độ chung, tỷ lệ chia năng lượng và pha STAR--RIS
để tối đa hóa tổng tốc độ, đồng thời thỏa ngân sách công suất, các ràng buộc vật lý của bề mặt và yêu
cầu QoS:
\begin{subequations}
\label{eq:bai-toan}
\begin{align}
\max_{p_c, \mathbf{p}_{\mathrm{priv}}, \boldsymbol{\eta}, \boldsymbol{\beta}^T,
\boldsymbol{\theta}^T, \boldsymbol{\theta}^R} \quad
& R_{\mathrm{sum}} = R_c + \sum_{k=1}^{K} R_{p,k} \\
\text{với ràng buộc} \quad
& p_c + \sum_{k=1}^{K} p_k = P_{\max}, \quad p_c \ge 0,\ p_k \ge 0, \label{eq:rb-cong-suat}\\
& \sum_{k=1}^{K} \eta_k = 1, \quad \eta_k \ge 0, \\
& 0 \le \beta^T_n \le 1, \quad \beta^R_n = 1 - \beta^T_n, \\
& -\pi \le \theta^T_n < \pi, \quad -\pi \le \theta^R_n < \pi, \label{eq:rb-pha}\\
& R_k \ge R_{\min}, \quad \forall k . \label{eq:rb-qos}
\end{align}
\end{subequations}

Các ràng buộc lần lượt là ngân sách công suất, bảo toàn tỷ lệ tốc độ chung, bảo toàn năng lượng của
bề mặt thụ động, miền pha và yêu cầu QoS. Bài toán~\eqref{eq:bai-toan} phi lồi vì ba lý do: các biến
công suất xuất hiện ở cả tử số và mẫu số của SINR; $R_c$ chứa phép lấy nhỏ nhất nên hàm mục tiêu
không trơn; và các biến pha nằm trong số mũ phức, chịu phép lấy bình phương độ lớn trong
$|h^{\mathrm{eff}}_k|^2$, tạo nhiều cực trị địa phương.

Trong thực nghiệm, không phương pháp nào giải trực tiếp bài toán có ràng buộc cứng~\eqref{eq:rb-qos}.
Các bộ giải AO tối ưu trực tiếp hàm đánh giá có phạt $r_{\mathrm{env}}$
trong~\eqref{eq:phan-thuong-moi-truong}; các thuật toán DRL học từ một phần thưởng huấn luyện có hệ số
phạt thích nghi (Mục~\ref{ssec:phan-thuong-huan-luyen}); còn khi kiểm thử, mọi phương pháp được so sánh
trên cùng tổng tốc độ và tỷ lệ đạt QoS. Ràng buộc QoS vì vậy được xử lý mềm.

\section{Phân tích cấu trúc SISO--RSMA}
\label{sec:phan-tich-cau-truc}

Mục này phân tích bản chất toán học và vật lý của hệ SISO kết hợp RSMA và STAR--RIS, làm cơ sở cho
việc thiết kế bộ giải mã hành động có cấu trúc thay vì để DRL tự học trên miền hành động nguyên bản.

\subsection{Trần tốc độ do nhiễu luồng riêng trong hệ SISO}
\label{ssec:nhieu-luong-rieng}

Trong hệ một anten phát, BS không có bậc tự do không gian để tạo búp sóng riêng cho từng luồng: mọi
luồng riêng đến UE $k$ qua cùng một hệ số $h^{\mathrm{eff}}_k$. Giả sử có $M \le K$ luồng riêng hoạt
động và chia đều công suất, $p_k = P_{\mathrm{priv}}/M$. Khi đó~\eqref{eq:sinr-rieng} trở thành
\begin{equation}
\gamma_{p,k} = \frac{g_k\, P_{\mathrm{priv}}/M}{g_k\, \frac{M-1}{M} P_{\mathrm{priv}} + \sigma^2}.
\label{eq:sinr-m-luong}
\end{equation}

Khi nhiễu luồng riêng lấn át tạp âm, tức $g_k \frac{M-1}{M} P_{\mathrm{priv}} \gg \sigma^2$, độ lợi
$g_k$ ở tử và mẫu triệt tiêu nhau:
\begin{equation}
\gamma_{p,k} \approx \frac{1}{M-1}, \qquad
R_{p,k} \approx \log_2\!\left(\frac{M}{M-1}\right), \qquad
\sum_{k=1}^{M} R_{p,k} \approx M \log_2\!\left(\frac{M}{M-1}\right).
\label{eq:xap-xi-nhieu}
\end{equation}

từ hai lên bốn luồng. Với $M = 4$, mỗi luồng riêng chỉ đạt khoảng $0,41$ bit/s/Hz, thấp hơn cả ngưỡng

luồng đó gần như chỉ còn tạp âm và
\begin{equation}
\gamma_{p,k^\star} \approx \frac{g_{k^\star} P_{\mathrm{priv}}}{\sigma^2},
\label{eq:tap-trung-cong-suat}
\end{equation}

họa hai trường hợp này: đường $M = 1$ tăng theo SNR, còn các đường $M = 2$ và $M = 4$ bão hòa ở trần
của mình.

hưởng lợi từ nhiều luồng riêng. Kết luận an toàn là bài toán có xu hướng ưu tiên phân bổ riêng tập
trung trong nhiều trạng thái kênh SISO, nên một tham số hóa có khả năng biểu diễn phân bổ gần
one-hot là hợp lý.

\subsection{Quan hệ bù trừ giữa luồng chung và luồng riêng}
\label{ssec:quan-he-chung-rieng}

Giả sử chỉ một UE $k^\star$ có luồng riêng, $p_{k^\star} = P_{\mathrm{priv}} = P_{\max} - p_c$ và
$p_j = 0$ với $j \neq k^\star$. Khi đó luồng riêng của $k^\star$ không chịu nhiễu từ luồng riêng khác:
\begin{equation}
R_{p,k^\star} = \log_2\!\left(1 + \frac{g_{k^\star}(P_{\max} - p_c)}{\sigma^2}\right),
\label{eq:toc-do-rieng-k-sao}
\end{equation}
và tốc độ giải mã luồng chung tại chính UE này theo~\eqref{eq:sinr-chung} là
\begin{equation}
R_{c,k^\star} = \log_2\!\left(\frac{g_{k^\star} P_{\max} + \sigma^2}
{g_{k^\star}(P_{\max} - p_c) + \sigma^2}\right).
\label{eq:toc-do-chung-k-sao}
\end{equation}
Cộng hai biểu thức, phần phụ thuộc $p_c$ triệt tiêu:
\begin{equation}
R_{c,k^\star} + R_{p,k^\star} = \log_2\!\left(1 + \frac{g_{k^\star} P_{\max}}{\sigma^2}\right).
\label{eq:tong-hai-thanh-phan}
\end{equation}

Vế phải là dung lượng của liên kết không nhiễu khi dồn toàn bộ $P_{\max}$ cho UE $k^\star$. Ngoài
ra, khi mọi UE ở miền tín hiệu trên tạp âm cao, tức $g_k(P_{\max} - p_c) \gg \sigma^2$ với mọi $k$, độ
lợi kênh trong~\eqref{eq:sinr-chung} tự giản ước và với $P_{\max} = 1$:
\begin{equation}
R_c \approx \log_2\!\left(1 + \frac{p_c}{1 - p_c}\right) = -\log_2(1 - p_c),
\label{eq:toc-do-chung-xap-xi}
\end{equation}
tức tốc độ luồng chung tăng nhanh khi $p_c$ tiến gần một. Tại UE này,
tăng $p_c$ làm phần tốc độ chung tăng đúng bằng phần tốc độ riêng giảm. Tuy nhiên, tốc độ luồng chung
thực tế là $R_c = \min_k R_{c,k} \le R_{c,k^\star}$, nên quan hệ bù trừ chỉ đúng chính xác khi
$k^\star$ cũng là UE nút cổ chai của luồng chung; với các UE kênh yếu hơn, $R_c$ nhỏ hơn và tổng tốc
độ giảm khi chuyển công suất sang luồng chung. Luồng chung vì vậy là công cụ phân phối dịch vụ có cái
giá: BS có thể chuyển công suất từ luồng riêng của UE mạnh sang luồng chung để chia lại cho các UE yếu
qua $C_k = \eta_k R_c$. Với cấu hình một luồng riêng như trên, mọi UE đạt QoS khi và chỉ khi có thể chọn
$\boldsymbol{\eta} \in \Delta_K$ sao cho $\eta_k R_c + R_{p,k} \ge R_{\min}$ với mọi $k$, tương đương
\begin{equation}
\sum_{k=1}^{K} \max\big(R_{\min} - R_{p,k},\, 0\big) \le R_c .
\label{eq:dieu-kien-qos}
\end{equation}
Điều kiện~\eqref{eq:dieu-kien-qos} cho thấy luồng chung chỉ bảo vệ được QoS khi $R_c$ đủ lớn; đây là cơ
sở để ưu tiên một vùng $p_c$ cao trong bộ giải mã ở Chương~\ref{chap:phuong-phap}, nhưng đó là một lựa
chọn thiết kế được kiểm chứng bằng thực nghiệm chứ không phải ràng buộc vật lý của RSMA.

\subsection{Trực giác căn pha của STAR--RIS}
\label{ssec:vai-tro-can-pha}

Sau phân bổ công suất, câu hỏi tiếp theo là điều khiển STAR--RIS thế nào để tăng độ lợi kênh hiệu dụng
$g_k = |h^{\mathrm{eff}}_k|^2$. Theo~\eqref{eq:kenh-hieu-dung}, $|h^{\mathrm{eff}}_k|$ lớn nhất khi mọi
thành phần qua STAR--RIS đến UE $k$ cùng pha với đường trực tiếp:
\begin{equation}
\angle\big(h^{*}_{r,k,n} g_n\big) + \theta^{s_k}_n = \angle h_{d,k}
\;\;\Rightarrow\;\;
\theta^{\mathrm{ideal}}_{n,k} = \angle h_{d,k} - \angle\big(h^{*}_{r,k,n} g_n\big).
\label{eq:pha-ly-tuong}
\end{equation}

Trong mạng nhiều người dùng, cùng một phần tử phải phục vụ đồng thời nhiều UE ở cùng một phía (ví dụ
UE$_1$ và UE$_3$ ở phía phản xạ). Vì kênh của các UE khác nhau, các pha lý tưởng
$\theta^{\mathrm{ideal}}_{n,1}$ và $\theta^{\mathrm{ideal}}_{n,3}$ xung đột, và một bề mặt không thể đặt
một pha làm hài lòng mọi UE cùng lúc.

Hình~\ref{fig:ch2-can-pha} minh họa hiệu ứng căn pha. Trong minh họa chỉ xét một UE $k^\star$ và tạm
bỏ đường trực tiếp. Ở khung bên trái, các thành phần ghép tầng có pha lệch nhau nên khi cộng tiếp nối,
chúng tạo thành một đường gấp khúc gần khép kín và véc-tơ tổng rất ngắn. Khi căn pha riêng cho UE này,
các thành phần ghép tầng cùng hướng và độ lớn của phần kênh qua STAR--RIS tăng, như ở khung bên phải.
Pha tham chiếu gộp nhiều UE dùng trong Chương~\ref{chap:phuong-phap} không bảo đảm các thành phần của
từng UE đều thẳng hàng.

Đây là điểm tựa cho vai trò của DRL: thay vì để mạng nơ-ron tự học căn pha trên toàn bộ vòng tròn

một góc hiệu chỉnh quanh tham chiếu đó nhằm cân bằng giữa các UE và mục tiêu RSMA.

\subsection{Hàm ý đối với thiết kế biểu diễn hành động}
\label{ssec:ham-y-thiet-ke}

Phân tích trên dẫn đến ba hàm ý. Thứ nhất, không gian công suất riêng cần dễ biểu diễn nghiệm tập
trung; softmax có độ sắc lớn phù hợp hơn việc buộc Actor tự đẩy đầu ra tới bão hòa. Thứ hai, tỷ lệ công
suất luồng chung nên được ưu tiên trong một vùng đủ cao để hỗ trợ QoS, như một lựa chọn thiết kế. Thứ
ba, pha nên có một điểm xuất phát dựa trên căn chỉnh kênh thay vì học toàn bộ $[-\pi, \pi)$ từ đầu. Ba
hàm ý này tạo nên bộ giải mã có cấu trúc ở Chương~\ref{chap:phuong-phap}; về mặt học máy, đây là cách
đưa thiên kiến quy nạp (inductive bias) vào chính không gian hành động.

\section{Giới hạn của mô hình}
\label{sec:gioi-han-mo-hinh}

Mô hình có một số giới hạn. Kênh Gaussian phức với hệ số tỷ lệ cố định không mô tả suy hao theo khoảng
cách, che khuất hay tương quan không gian, và diện tích hiệu dụng của bề mặt tăng theo $N$. CSI hoàn hảo
bỏ qua sai số ước lượng. SIC lý tưởng bỏ qua lan truyền lỗi. Pha truyền và phản xạ được điều khiển độc
lập, trong khi STAR--RIS thụ động thực tế có ràng buộc ghép pha~\cite{Liu2021STARCoupledPhase,
Wang2022STAROptimization}. Số UE cố định bằng bốn nên kết quả chưa chứng minh khả năng tổng quát theo số
người dùng. Những giới hạn này là phạm vi của nghiên cứu, không phải lỗi của thuật toán.

\section{Kết luận chương}
\label{sec:ch2-ket-luan}

Chương~\ref{chap:mo-hinh-he-thong} đã xây dựng mô hình kênh, mô hình STAR--RIS ở chế độ ES và mô hình
RSMA một lớp nhất quán với mã nguồn, rồi phát biểu bài toán tối ưu tổng tốc độ có ràng buộc QoS. Phân
tích cấu trúc cho thấy trong hệ SISO, nhiễu giữa nhiều luồng riêng tạo ra trần tốc độ gần như không phụ
thuộc kênh; luồng chung là công cụ phân phối dịch vụ có điều kiện; còn căn pha STAR--RIS có thể tăng độ
lợi kênh nhưng phải dung hòa giữa nhiều UE. Các hàm ý này là cơ sở để thiết kế bộ giải mã hành động có
cấu trúc vật lý ở Chương~\ref{chap:phuong-phap}.
