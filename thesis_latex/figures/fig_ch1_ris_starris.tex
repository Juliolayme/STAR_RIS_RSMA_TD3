% ---------- macro cuc bo ----------
% khoi phan tu: x, z, mau
\newcommand{\rsCube}[3]{%
  \begin{scope}[shift={(#1,#2)}]
    \fill[#3!85!black] (0,0) -- ++(0.15,0.09) -- ++(0,0.30) -- ++(-0.15,-0.09) -- cycle;
    \fill[#3!40] (0,0.30) -- ++(0.15,0.09) -- ++(-0.26,0) -- ++(-0.15,-0.09) -- cycle;
    \fill[#3!65] (-0.26,0) rectangle (0,0.30);
    \draw[#3!50!black,line width=0.3pt] (-0.26,0) rectangle (0,0.30)
      (0,0) -- ++(0.15,0.09) -- ++(0,0.30) -- ++(-0.26,0) -- (-0.26,0.30) (0,0.30) -- ++(0.15,0.09);
  \end{scope}}
% be mat (tam phang dung) + luoi 3x4 khoi: goc truoc-duoi x, mau khoi
\newcommand{\rsSurface}[2]{%
  \fill[gray!25,opacity=0.85] (#1,-0.55) -- ++(0.99,0.594) -- ++(0,2.75) -- ++(-0.99,-0.594) -- cycle;
  \draw[gray!60] (#1,-0.55) -- ++(0.99,0.594) -- ++(0,2.75) -- ++(-0.99,-0.594) -- cycle;
  \foreach \j in {0,1,2}{\foreach \i in {0,1,2,3}{
     \rsCube{#1+0.33+0.255*\j}{-0.30+0.153*\j+0.62*\i}{#2}}}}
% nguoi dung: x, y, mau, nhan
\newcommand{\rsUE}[4]{%
  \fill[#3] (#1,#2+0.33) circle (0.12);
  \fill[#3] (#1-0.22,#2) arc (180:0:0.22 and 0.24) -- cycle;
  \node[font=\scriptsize,anchor=north] at (#1,#2-0.02) {#4};}
% nguoi dung khong duoc phuc vu (vien dut, xam)
\newcommand{\rsUEoff}[3]{%
  \draw[gray!60,dashed] (#1,#2+0.33) circle (0.12);
  \draw[gray!60,dashed] (#1-0.22,#2) arc (180:0:0.22 and 0.24) -- cycle;
  \node[font=\scriptsize,anchor=north,gray!60] at (#1,#2-0.02) {#3};}
% tram goc nho: x, y
\newcommand{\rsBS}[2]{%
  \begin{scope}[shift={(#1,#2)}]
    \fill[gray!12] (-0.30,0) rectangle (0.30,0.80);
    \draw (-0.30,0) -- (-0.30,0.80) (0.30,0) -- (0.30,0.80);
    \draw[fill=gray!20] (0,0.80) ellipse (0.30 and 0.09);
    \draw (-0.30,0) arc (180:360:0.30 and 0.09);
    \foreach \i in {0,1}{\foreach \j in {0,1}{\fill[gray!55] (-0.18+\i*0.22,0.15+\j*0.30) rectangle ++(0.14,0.20);}}
    \draw[line width=1.3pt] (0,-0.09) -- (0,-0.45);
    \node[anchor=south] at (0,0.90) {BS};
  \end{scope}}
